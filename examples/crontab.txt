# and are emailed like web runs. Edit with: crontab -e
#

0 7 * * 1,3,5  /usr/bin/env node $HOME/claude-runner-ui/bin/run-job.mjs auto "run the daily job" cron >> $HOME/claude-runner-ui/runs/cron.log 2>&1
